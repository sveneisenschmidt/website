{{/* Was ein Beitrag ist, in einer Zeile — und die Angabe, die nur dieser Typ hat. {{ partial "article-type.html" (dict "page" .) }} mit Typ {{ partial "article-type.html" (dict "page" . "type" false) }} ohne Ohne den Typ auf einer Beitragsseite: dort steht er schon als Label über dem Titel, und dasselbe Wort zweimal sagt nichts doppelt. Gehört ein Beitrag zu einer Reise, nennt die Zeile die Stelle in der Folge statt des Typs — „Summer 2026, day 3" ist mehr wert als „Trip". Der Tag wird aus der Reihenfolge nach Datum gezählt, nicht aus dem Slug: ein Slug kann lügen, ein Datum nicht. */}} {{- $page := .page -}} {{- $withType := true -}} {{- if eq .type false }}{{ $withType = false }}{{ end -}} {{- with $page -}} {{- with .Params.trip -}} {{- $days := sort (where site.RegularPages "Params.trip" .) "Date" -}} {{- $name := . | replaceRE `-` " " | title -}} {{- range $i, $d := $days -}} {{- if eq $d.RelPermalink $page.RelPermalink -}} {{- printf "%s, day %d of %d" $name (add $i 1) (len $days) -}} {{- end -}} {{- end -}} {{- else -}} {{- $type := .Params.type | default "" -}} {{- if eq $type "essay" -}} {{- if $withType }}Essay, {{ end }}{{ .ReadingTime }} min {{- else if and $withType (eq $type "note") -}} Note {{- else if and $withType (eq $type "photo") -}} Photo {{- end -}} {{- end -}} {{- end -}}
